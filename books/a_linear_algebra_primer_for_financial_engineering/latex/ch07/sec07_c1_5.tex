\section{Covariance and correlation matrix estimation from time series data}

Let $X_1, X_2, \ldots, X_n$ be random variables given by time series data at $N$
data points $t_i$, $i = 1:N$, with $N > n$. Denote by $T_{X_k}$ the column vector of
the time series data for the random variable $X_k$, for $k = 1:n$, i.e.,
\begin{equation}
T_{X_k} \;=\; (X_k(t_i))_{i=1:N}, \quad \forall\, k = 1:n,
\end{equation}
and denote by
\begin{equation}
T_{\bfX} \;=\; \mathrm{col}\,(T_{X_k})_{k=1:n}
\end{equation}
the $N \times n$ matrix of time series data for the $n$ random variables.

The sample mean $\widehat{\mu}_{X_k}$ of $X_k$, the unbiased sample variance
$\widehat{\sigma}_k^2$ of $X_k$, and the unbiased sample covariance
$\widehat{\cov}(X_j,X_k)$ and sample correlation $\widehat{\corr}(X_j,X_k)$ of $X_j$
and $X_k$ are given, respectively, by
\begin{align}
\widehat{\mu}_{X_k} &= \frac{1}{N}\sum_{i=1}^N X_k(t_i), \quad \forall\, k = 1:n; \\
\widehat{\sigma}_k^2 &= \frac{1}{N-1}\sum_{i=1}^N \left(X_k(t_i)-\widehat{\mu}_{X_k}\right)^2, \quad \forall\, k = 1:n; \\
\widehat{\cov}(X_j,X_k) &= \frac{1}{N-1}\sum_{i=1}^N \left(X_j(t_i)-\widehat{\mu}_{X_j}\right)\left(X_k(t_i)-\widehat{\mu}_{X_k}\right), \quad \forall\, 1\le j,k\le n; \\
\widehat{\corr}(X_j,X_k) &= \frac{\widehat{\cov}(X_j,X_k)}{\widehat{\sigma}_j\widehat{\sigma}_k}, \quad \forall\, 1\le j,k\le n.
\end{align}

The sample covariance matrix $\widehat{\Sigma}_{\bfX}$ and the sample correlation
matrix $\widehat{\Omega}_{\bfX}$ of the random variables $X_1, X_2, \ldots, X_n$ are
the $n \times n$ matrices given by
\begin{align}
\widehat{\Sigma}_{\bfX}(j,k) &= \widehat{\cov}(X_j,X_k), \quad \forall\, 1\le j,k\le n; \\
\widehat{\Omega}_{\bfX}(j,k) &= \widehat{\corr}(X_j,X_k), \quad \forall\, 1\le j,k\le n.
\end{align}

The following connections between the sample covariance matrix
$\widehat{\Sigma}_{\bfX}$ and the sample correlation matrix $\widehat{\Omega}_{\bfX}$
can be established as in Lemma 7.1:
\begin{align}
\widehat{\Sigma}_{\bfX} &= \widehat{D}_{\sigma_{\bfX}}\widehat{\Omega}_{\bfX}\widehat{D}_{\sigma_{\bfX}}; \\
\widehat{\Omega}_{\bfX} &= (\widehat{D}_{\sigma_{\bfX}})^{-1}\widehat{\Sigma}_{\bfX}(\widehat{D}_{\sigma_{\bfX}})^{-1},
\end{align}
where $\widehat{D}_{\sigma_{\bfX}} = \diag(\widehat{\sigma}_k)_{k=1:n}$ and
$(\widehat{D}_{\sigma_{\bfX}})^{-1} =
\diag\!\left(\frac{1}{\widehat{\sigma}_k}\right)_{k=1:n}$. Note that (7.44) and
(7.45) can also be written as follows:
\begin{align}
\widehat{\Sigma}_{\bfX} &= \diag(\widehat{\sigma}_k)_{k=1:n}\ \widehat{\Omega}_{\bfX}\ \diag(\widehat{\sigma}_k)_{k=1:n}; \\
\widehat{\Omega}_{\bfX} &= \diag\!\left(\frac{1}{\widehat{\sigma}_k}\right)_{k=1:n} \widehat{\Sigma}_{\bfX} \diag\!\left(\frac{1}{\widehat{\sigma}_k}\right)_{k=1:n}.
\end{align}
