\textbf{I-5.}
\[
  \bfE = \frac{2y}{4\pi\epszero\,(1+y^2)^{3/2}}\,\bfj

\medskip
\textbf{I-6.} Field lines ($c$ = arbitrary constant):
\begin{enumerate}[label=(\alph*)]
  \item $x^2 - y^2 = c$
  \item $y = x + c$
  \item $xy = c$
  \item $y = c$
  \item $x = c$
  \item $x^2 - y^2 = c$
  \item $y = 2x^2 + c$
  \item $y = c\,e^x$
\end{enumerate}
